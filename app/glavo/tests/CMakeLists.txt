cmake_minimum_required(VERSION 3.16)
project(glavo_cabin_exchange_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../../common ${CMAKE_CURRENT_BINARY_DIR}/common)
add_executable(cabin_visual_exchange_tests cabin_visual_exchange_tests.cpp)
target_compile_features(cabin_visual_exchange_tests PRIVATE cxx_std_17)
target_include_directories(cabin_visual_exchange_tests PRIVATE ../lib ../../visual/lib)
target_link_libraries(cabin_visual_exchange_tests PRIVATE upc::common Threads::Threads)
target_compile_options(cabin_visual_exchange_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME cabin_visual_exchange_tests COMMAND cabin_visual_exchange_tests)
set_tests_properties(cabin_visual_exchange_tests PROPERTIES TIMEOUT 15)
